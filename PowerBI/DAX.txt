-- DAX Measures for Data Quality Accuracy Dashboard
-- Tableau de Bord de Qualité des Données

-- ============================================
-- GLOBAL ACCURACY SCORE
-- ============================================
Accuracy Score Global = 
VAR TotalRecords = 
    COUNTROWS(Dim_Client) + 
    COUNTROWS(Dim_Compte) + 
    COUNTROWS(Dim_Transaction) + 
    COUNTROWS(Dim_Credit)

VAR TotalIssues = COUNTROWS(Fact_Accuracy)

VAR AccuracyScore = 
    IF(
        TotalRecords > 0,
        DIVIDE(TotalRecords - TotalIssues, TotalRecords) * 100,
        0
    )

RETURN
    FORMAT(AccuracyScore, "0.00") & "%"

-- SLA COMPLIANCE (>= 95%)

VAR TotalIssues = COUNTROWS(Fact_Accuracy)

-- ============================================
-- VALID RECORDS

-- ============================================

-- ============================================
Invalid Records = COUNTROWS(Fact_Accuracy)

-- ============================================
-- AGENCY ACCURACY % (For Agence_Name Histogram)
-- ============================================
Agency Accuracy % = 
VAR TotalIssues = COUNTROWS(Fact_Accuracy)

VAR IssuesByAgency = 
    COUNTROWS(
        FILTER(
            Fact_Accuracy, 
            Fact_Accuracy[Agence_Key] = MAX(Dim_Agence[Agence_Key])
        )
    )

VAR TotalRecordsByAgency = 
    DIVIDE(IssuesByAgency, TotalIssues) * 
    (COUNTROWS(Dim_Client) + COUNTROWS(Dim_Compte) + COUNTROWS(Dim_Transaction) + COUNTROWS(Dim_Credit))

VAR AccuracyScore = 
    IF(
        TotalRecordsByAgency > 0,
        DIVIDE(TotalRecordsByAgency - IssuesByAgency, TotalRecordsByAgency) * 100,
        100
    )

RETURN
    AccuracyScore

-- ============================================
-- AGENCY ACCURACY % FOR CLIENT TABLE
-- ============================================
Agency Client Accuracy % =
VAR TotalClientIssues = COUNTROWS(FILTER(Fact_Accuracy, Fact_Accuracy[Table_Name] = "Client"))

VAR ClientIssuesByAgency = COUNTROWS(
    FILTER(
        Fact_Accuracy,
        Fact_Accuracy[Table_Name] = "Client" &&
        Fact_Accuracy[Agence_Key] = MAX(Dim_Agence[Agence_Key])
    )
)

VAR TotalClients = COUNTROWS(Dim_Client)

VAR ClientsByAgency =
    DIVIDE(ClientIssuesByAgency, TotalClientIssues) * TotalClients

VAR AccuracyScore =
    IF(
        ClientsByAgency > 0,
        DIVIDE(ClientsByAgency - ClientIssuesByAgency, ClientsByAgency) * 100,
        100
    )

RETURN
    AccuracyScore

    AccuracyScore
